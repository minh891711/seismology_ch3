讀取完整摘要檔案 summary_path = "summaries/Seismology_Chapter3_Full_Summary.md" if not os.path.exists(summary_path): print(f"找不到檔案: {summary_path}") return with open(summary_path, "r", encoding="utf-8") as f: full_content = f.read() #
